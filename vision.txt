import cv2
import numpy as np
import math
import config


def calibrate_and_warp(image_path):
    """
    Detects 4 ArUco markers to correct perspective.
    Robust version: checks all dictionaries and handles poor lighting.
    """
    W, H, SCALE = 150.0, 237.0, 5.0

    img = cv2.imread(image_path)
    if img is None: raise ValueError("Image file not found.")

    if len(img.shape) == 3:
        gray_original = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    else:
        gray_original = img

    params = cv2.aruco.DetectorParameters()
    # Увеличаваме параметрите, за да хваща маркери дори при висока резолюция и лош контраст
    params.adaptiveThreshWinSizeMax = 53
    params.adaptiveThreshWinSizeStep = 10

    # Списък с ВСИЧКИ стандартни масиви, за да няма значение какъв маркер си принтирал
    dicts = [
        cv2.aruco.DICT_4X4_50, cv2.aruco.DICT_4X4_250,
        cv2.aruco.DICT_5X5_50, cv2.aruco.DICT_5X5_250,  # Точно тези липсваха за новата ти снимка
        cv2.aruco.DICT_6X6_50, cv2.aruco.DICT_6X6_250,
        cv2.aruco.DICT_7X7_50, cv2.aruco.DICT_7X7_250,
        cv2.aruco.DICT_ARUCO_ORIGINAL
    ]

    # 3 варианта за осветление: 1. Нормално, 2. Подсилен контраст, 3. Тъмно
    clahe = cv2.createCLAHE(clipLimit=3.0, tileGridSize=(8, 8))
    preprocessing_steps = [
        gray_original,
        clahe.apply(gray_original),
        cv2.GaussianBlur(gray_original, (5, 5), 0)
    ]

    corners, ids = None, None

    # Търсим агресивно: пробваме всяко осветление с всеки възможен тип маркер
    for gray in preprocessing_steps:
        for d_id in dicts:
            detector = cv2.aruco.ArucoDetector(cv2.aruco.getPredefinedDictionary(d_id), params)
            c, i, _ = detector.detectMarkers(gray)

            # Ако намери поне 4 маркера, спираме търсенето
            if i is not None and len(i) >= 4:
                corners, ids = c[:4], i[:4]  # Взимаме само първите 4 (ако има "шум")
                break
        if ids is not None and len(ids) == 4:
            break

    if ids is None or len(ids) < 4:
        raise ValueError("Markers not found! Ensure 4 markers are visible and clear.")

    # Sort markers: Top-Left, Top-Right, Bottom-Right, Bottom-Left
    centers = np.array([np.mean(c[0], axis=0) for c in corners])
    s = centers.sum(axis=1)
    diff = np.diff(centers, axis=1)

    src_pts = np.array([centers[np.argmin(s)], centers[np.argmin(diff)],
                        centers[np.argmax(s)], centers[np.argmax(diff)]], dtype="float32")

    dw, dh = int(W * SCALE), int(H * SCALE)
    dst_pts = np.array([[0, 0], [dw - 1, 0], [dw - 1, dh - 1], [0, dh - 1]], dtype="float32")
    matrix = cv2.getPerspectiveTransform(src_pts, dst_pts)

    return cv2.warpPerspective(img, matrix, (dw, dh))


def analyze_part(warped):
    """
    Advanced dual-mask analysis with Shadow Rejection and Perfect Geometry formatting.
    """
    if warped is None: return None, [], 0, 0, 'rectangle', 0, 0, None

    labeled_image = warped.copy()
    if len(warped.shape) == 3:
        gray = cv2.cvtColor(warped, cv2.COLOR_BGR2GRAY)
    else:
        gray = warped

    # 1. Equalize lighting
    clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8, 8))
    gray = clahe.apply(gray)
    blurred = cv2.GaussianBlur(gray, (7, 7), 0)

    # 2. Get standard Otsu threshold
    otsu_val, _ = cv2.threshold(blurred, 0, 255, cv2.THRESH_BINARY_INV + cv2.THRESH_OTSU)

    # --- STRATEGY 1: SHADOW REJECTION FOR OUTER CONTOUR ---
    # Make the threshold 25% stricter! This drops the lighter shadow, keeping only the dark metal.
    strict_val = otsu_val * 0.75
    _, mask_outer = cv2.threshold(blurred, strict_val, 255, cv2.THRESH_BINARY_INV)

    # Morphological CLOSE to completely fill holes and bridge metallic glares
    kernel_solid = cv2.getStructuringElement(cv2.MORPH_RECT, (25, 25))
    mask_outer = cv2.morphologyEx(mask_outer, cv2.MORPH_CLOSE, kernel_solid)

    contours_outer, _ = cv2.findContours(mask_outer, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    if not contours_outer: return None, [], 0, 0, 'rectangle', 0, 0, None

    valid_outer = [c for c in contours_outer if cv2.contourArea(c) > 3000]
    if not valid_outer: return None, [], 0, 0, 'rectangle', 0, 0, None

    outer_cnt = max(valid_outer, key=cv2.contourArea)

    # 3. Shape analysis and mathematical perfection
    M = cv2.moments(outer_cnt)
    if M['m00'] == 0: return None, [], 0, 0, 'rectangle', 0, 0, None
    pcx, pcy = M['m10'] / M['m00'], M['m01'] / M['m00']

    peri = cv2.arcLength(outer_cnt, True)
    circ = (4 * math.pi * M['m00']) / (peri ** 2) if peri > 0 else 0
    shape_type = 'circle' if circ > 0.82 else 'rectangle'

    # Get mathematically perfect bounding box (removes any wobbly edges)
    rect = cv2.minAreaRect(outer_cnt)
    (_, _), (w_px, h_px), _ = rect

    if len(labeled_image.shape) == 2:
        labeled_image = cv2.cvtColor(labeled_image, cv2.COLOR_GRAY2BGR)

    # Draw Perfect Geometry
    if shape_type == 'rectangle':
        box = cv2.boxPoints(rect)
        perfect_cnt = np.int32(box)
        cv2.drawContours(labeled_image, [perfect_cnt], 0, (0, 255, 0), 2)
        test_cnt = perfect_cnt  # Use perfect box for hole intersection test
    else:
        radius = (w_px + h_px) / 4.0
        cv2.circle(labeled_image, (int(pcx), int(pcy)), int(radius), (0, 255, 0), 2)
        test_cnt = outer_cnt  # Fallback

    # --- STRATEGY 2: CLEAN MASK FOR INTERNAL HOLES ---
    # We use the standard Otsu here so we don't shrink the actual holes
    _, mask_holes = cv2.threshold(blurred, otsu_val, 255, cv2.THRESH_BINARY_INV)
    kernel_clean = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (3, 3))
    mask_holes = cv2.morphologyEx(mask_holes, cv2.MORPH_OPEN, kernel_clean)

    contours_all, hierarchy = cv2.findContours(mask_holes, cv2.RETR_TREE, cv2.CHAIN_APPROX_SIMPLE)

    features_px = []
    fid = 1

    if hierarchy is not None:
        hierarchy = hierarchy[0]
        for i, cnt in enumerate(contours_all):
            m = cv2.moments(cnt)
            if m['m00'] < 50: continue
            if m['m00'] > M['m00'] * 0.5: continue

            fx, fy = m['m10'] / m['m00'], m['m01'] / m['m00']

            # Check if feature is strictly INSIDE the perfect contour
            if cv2.pointPolygonTest(test_cnt, (fx, fy), False) >= 0:
                if hierarchy[i][3] != -1:
                    f_area = m['m00']
                    f_peri = cv2.arcLength(cnt, True)
                    f_circ = (4 * math.pi * f_area) / (f_peri ** 2) if f_peri > 0 else 0
                    f_rect = cv2.minAreaRect(cnt)
                    f_ar = min(f_rect[1]) / max(f_rect[1]) if max(f_rect[1]) > 0 else 0

                    is_hole = (f_circ > 0.45) or (f_ar > 0.6)

                    # Prevent duplicates
                    is_duplicate = False
                    for feat in features_px:
                        if math.hypot(feat['px_x'] - fx, feat['px_y'] - fy) < 15:
                            is_duplicate = True
                            break

                    if not is_duplicate:
                        features_px.append({
                            'id': fid, 'px_x': fx, 'px_y': fy, 'px_area': f_area, 'is_hole': is_hole
                        })
                        cv2.drawContours(labeled_image, [cnt], -1, (255, 0, 0), 1)
                        cv2.putText(labeled_image, f"#{fid}", (int(fx) - 10, int(fy) + 10),
                                    cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 0, 255), 2)
                        fid += 1

    return outer_cnt, features_px, w_px, h_px, shape_type, pcx, pcy, labeled_image
