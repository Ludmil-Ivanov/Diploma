import os
import sys
import cv2
import math
import tkinter as tk
from tkinter import filedialog, messagebox, simpledialog

import vision
import gcode_generator
import cam_logic
import verification_ui
import config


def main():
    root = tk.Tk()
    root.withdraw()

    # 1. Warp Perspective
    file_path = filedialog.askopenfilename(title="Select Part Image")
    if not file_path:
        return

    try:
        warped_img = vision.calibrate_and_warp(file_path)
    except Exception as e:
        messagebox.showerror("Error", str(e))
        return

    # 2. Analyze Part (Expect 8 return values)
    outer, f_px, w_px, h_px, shape, pcx, pcy, labeled_img = vision.analyze_part(warped_img)
    if outer is None:
        messagebox.showerror("Error", "No part detected.")
        return

    # 3. Interactive Calibration
    scale_x, scale_y = 1.0, 1.0
    real_w, real_h = 0.0, 0.0
    bx, by, bw, bh = cv2.boundingRect(outer)

    if shape == 'circle':
        real_dia = simpledialog.askfloat(title="Input", prompt="Detected: CIRCLE\nEnter REAL Diameter (mm):")
        if not real_dia:
            sys.exit()
        scale_x = bw / real_dia
        scale_y = bh / real_dia
        real_w, real_h = real_dia, real_dia
    else:
        real_w = simpledialog.askfloat(title="Input", prompt="Detected: RECTANGLE\nEnter REAL Width (X mm):")
        real_h = simpledialog.askfloat(title="Input", prompt="Enter REAL Height (Y mm):")
        if not real_w or not real_h:
            sys.exit()
        scale_x = bw / real_w
        scale_y = bh / real_h

    # 4. Feature Scaling
    final_features = []
    for f in f_px:
        cnc_x = (f['px_x'] - pcx) / scale_x
        cnc_y = -(f['px_y'] - pcy) / scale_y
        dia_mm = (2 * math.sqrt(f['px_area'] / math.pi)) / ((scale_x + scale_y) / 2.0)

        final_features.append({
            'id': f['id'], 'shape': 'hole' if f['is_hole'] else 'pocket',
            'cnc_x': cnc_x, 'cnc_y': cnc_y, 'raw_dia': dia_mm,
            'cx_px': f['px_x'], 'cy_px': f['px_y']
        })

    # --- 4.5 AUTO-ALIGNMENT (Tolerance Snapping) ---
    # If two holes are within 1.5 mm on X or Y, align them perfectly.
    TOLERANCE_MM = 1.5

    # Align X axis (Vertical line)
    for i in range(len(final_features)):
        for j in range(i + 1, len(final_features)):
            if abs(final_features[i]['cnc_x'] - final_features[j]['cnc_x']) <= TOLERANCE_MM:
                avg_x = (final_features[i]['cnc_x'] + final_features[j]['cnc_x']) / 2.0
                final_features[i]['cnc_x'] = avg_x
                final_features[j]['cnc_x'] = avg_x

    # Align Y axis (Horizontal line)
    for i in range(len(final_features)):
        for j in range(i + 1, len(final_features)):
            if abs(final_features[i]['cnc_y'] - final_features[j]['cnc_y']) <= TOLERANCE_MM:
                avg_y = (final_features[i]['cnc_y'] + final_features[j]['cnc_y']) / 2.0
                final_features[i]['cnc_y'] = avg_y
                final_features[j]['cnc_y'] = avg_y

    # 5. Verification Table
    if not verification_ui.verify_part_size(real_w, real_h):
        sys.exit()
    if final_features:
        final_features = verification_ui.verify_features(final_features, labeled_img)
    else:
        messagebox.showinfo("Info", "Solid part. No holes detected.")

    # 6. Tooling and Export
    drills = cam_logic.load_tools(config.DRILLS_DIR)
    mills = cam_logic.load_tools(config.MILLS_DIR)
    planned, needs_mill, report = cam_logic.plan_operations(final_features, drills)

    skip_outer = messagebox.askyesno("Strategy", "Skip outer milling?")

    selected_mill = None
    if needs_mill or not skip_outer:
        while True:
            m_dia = simpledialog.askfloat("Tooling", f"Available Mills: {mills}\nEnter End Mill Ø:")
            if not m_dia:
                sys.exit()
            if m_dia in mills:
                selected_mill = m_dia
                break
            messagebox.showwarning("Error", "Tool not in library.")

    out_path = gcode_generator.export_haas(outer, planned, selected_mill, skip_outer, scale_x, pcx, pcy, shape)
    messagebox.showinfo("Success", f"G-code saved: {out_path}")


if __name__ == "__main__":
    main()
