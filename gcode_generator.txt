import os
import cv2
import numpy as np


def export_haas(outer_contour, features, mill_tool, skip_outer, pixels_per_mm, center_x_px, center_y_px, outer_shape):
    """
    Universal G-code generator.
    Saves to Output_nc and chooses strategy based on outer_shape.
    """
    output_dir = "Output_nc"
    if not os.path.exists(output_dir):
        os.makedirs(output_dir)

    file_name = "haas_vf2_output.nc"
    out_path = os.path.join(output_dir, file_name)

    gcode = ["%", "O4000 (HAAS UNIVERSAL)", "G21 G90 G40 G80 G17 G54", "G00 Z50.0"]
    tool_num = 1

    # --- 1. DRILLING ---
    holes = [f for f in features if f['shape'] == 'hole']
    if holes:
        gcode.append("(--- DRILLING ---)")
        gcode.append(f"T{tool_num} M06\nS2500 M03\nG43 H{tool_num} Z50.0 M08")
        first = True
        for h in holes:
            if first:
                gcode.append(f"G00 X{h['cnc_x']:.3f} Y{h['cnc_y']:.3f}\nG81 Z-5.0 R2.0 F200.")
                first = False
            else:
                gcode.append(f"X{h['cnc_x']:.3f} Y{h['cnc_y']:.3f}")
        gcode.append("G80 M09\nG00 Z50.0")
        tool_num += 1

    # --- 2. OUTER PROFILE ---
    if not skip_outer and outer_contour is not None:
        gcode.append(f"(--- OUTER PROFILE: {outer_shape.upper()} ---)")
        if tool_num > 1: gcode.append(f"T{tool_num} M06")
        gcode.append(f"S4500 M03\nG43 H{tool_num} Z50.0 M08")

        tool_rad = (mill_tool / 2.0) if mill_tool else 0
        rect = cv2.minAreaRect(outer_contour)

        if outer_shape == 'circle':
            # Radius calculation
            real_radius = (max(rect[1]) / pixels_per_mm) / 2.0
            path_radius = real_radius + tool_rad
            gcode.append(f"G00 X{path_radius:.3f} Y0.0 (Start at side)")
            gcode.append("G01 Z-5.0 F200.")
            gcode.append(f"G03 X{path_radius:.3f} Y0.0 I{-path_radius:.3f} J0.0 F400. (Full Circle)")

        else:
            # Rectangle strategy
            box = cv2.boxPoints(rect)
            cnc_pts = []
            for pt in box:
                mx = (pt[0] - center_x_px) / pixels_per_mm
                my = -((pt[1] - center_y_px) / pixels_per_mm)
                # Offset away from center
                mx += tool_rad if mx > 0 else -tool_rad
                my += tool_rad if my > 0 else -tool_rad
                cnc_pts.append((mx, my))

            gcode.append(f"G00 X{cnc_pts[0][0]:.3f} Y{cnc_pts[0][1]:.3f}")
            gcode.append("G01 Z-5.0 F200.")
            for i in range(1, 4):
                gcode.append(f"G01 X{cnc_pts[i][0]:.3f} Y{cnc_pts[i][1]:.3f} F600.")
            gcode.append(f"G01 X{cnc_pts[0][0]:.3f} Y{cnc_pts[0][1]:.3f} (Close)")

        gcode.append("G01 Z5.0 F1000.\nM09")

    gcode.extend(["G00 Z50.0 M05", "G53 X0 Y0", "M30", "%"])

    with open(out_path, "w") as f:
        f.write("\n".join(gcode))
    return out_path
